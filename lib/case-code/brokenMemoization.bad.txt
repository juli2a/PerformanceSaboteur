// components/dashboard/MicroCard.tsx
const MicroCard = memo(function MicroCard({
  card,
  threshold,
}: Props) {
  const sparklineData = useMemo(
    () => processSparklineHistory(card.rawHistory),
    // card is a new object every render, this cache
    // never hits
    [card],
  );
  return (
    <MicroCardView
      sparklineData={sparklineData}
      id={card.id}
      title={card.meta.title}
      sku={card.meta.sku}
      marginality={card.marginality}
      value={formatCurrency(card.metrics.currentValue)}
      rating={card.metrics.rating}
      sparklineData={sparklineData}
      lowMargin={card.marginality < threshold}
    />
  );
});
export default MicroCard;

// lib/utils/sparkline-processing.ts
// Cleaning + smoothing + downsampling a year of daily
// readings
export function processSparklineHistory(
  raw: number[],
): number[] {
  const cleaned = removeOutliersIterative(raw);
  const smoothed = movingAverage(cleaned, 7);
  return downsampleTo(smoothed, 7);
}
